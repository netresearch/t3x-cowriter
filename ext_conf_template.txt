# cat=Translation; type=boolean; label=Revise DeepL translations with a language model: After a specialized translator such as DeepL translated the selected text, a language model revises the result for fluency, terminology and tone. # cat=Image generation; type=boolean; label=Generate images from a prompt in the editor: Adds the toolbar button "Cowriter - Generate image" to the rich text editor. An editor describes the image, nr-llm's OpenAI image service generates it, and the file is stored in the editor's default upload folder with the editor's file permissions. Needs an OpenAI API key in the nr-llm extension configuration. Each image counts against the editor's nr-llm budget.

# cat=Inline suggestions; type=boolean; label=Suggest a continuation while typing in the rich text editor: When an editor stops typing at the end of a paragraph, the language model proposes how the sentence goes on; it appears as grey text after the cursor, Tab inserts it, any other key discards it. Each editor can switch it off with the toolbar button "Cowriter - Inline suggestions". Every suggestion is a request to the language model and counts against the editor's nr-llm budget, also when the editor does not take it; at most one every five seconds per rich text field and 12 per minute per backend user.
inlineSuggestions.enable = 0

# cat=Inline suggestions; type=int+; label=Pause before a suggestion (milliseconds): How long the editor waits after the last key before it asks for a suggestion, from 800 to 3000.

# cat=Inline suggestions; type=int+; label=Minimum text (characters): How many characters the paragraph needs before suggestions start, from 10 to 500.
inlineSuggestions.minChars = 30

# cat=Alternative presentation; type=boolean; label=Present a text element as tabs or as an accordion: Adds the button "Present as tabs or accordion" next to the rich text of a text element (Text, Text & Images, Text & Media). It cuts the saved text into sections and, after the editor confirmed the preview, creates a new Tabs or Accordion element of bootstrap_package directly after the original, with one item per section. The original element stays unchanged; the editor deletes or hides it. A text with h2 or h3 headings is cut at them without a request to the language model; a text without them is split by the language model, which counts against the editor's nr-llm budget. Needs bootstrap_package; the button is shown only to editors who may create these elements on the page.
